# Cross-compiles the bench harness (static libparakeet + ggml) for iOS / Android.
cmake_minimum_required(VERSION 3.16)
project(parakeet_mobile_bench CXX C)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(PARAKEET_WITH_CED OFF CACHE BOOL "" FORCE)
set(PARAKEET_WITH_VOICEDETECT OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. parakeet)
add_executable(bench ../bench.cpp)
target_include_directories(bench PRIVATE ../../src ../../include)
target_link_libraries(bench PRIVATE parakeet)
